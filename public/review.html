Review Queue

Human-in-the-loop candidate confirmation. High-confidence items can be accepted directly; low-confidence items can be reassigned or flagged as new baseline activities.

async function load() { const status = document.getElementById('statusFilter').value; const el = document.getElementById('queue'); try { const items = await api('/review' + (status ? `?status=${status}` : '')); if (!items.length) { el.innerHTML = '
No queue items matching current status filter.
'; return; } el.innerHTML = items.map(item => { const matchedActivityId = item.activity_id; return `
${escapeHtml(item.activity_description)} ${escapeHtml(item.status)} ${escapeHtml(item.event_type)} Extracted Date: ${escapeHtml(item.extracted_date)}
${escapeHtml(item.discipline)}
Source: ${escapeHtml(item.source_type)} • Submitted by ${escapeHtml(item.submitted_by_name || 'unknown')}
Candidate Baseline Match Candidates (Click to Reassign):
${item.candidates.length ? item.candidates.map(c => { const isSelected = c.activity_id === matchedActivityId; const scorePct = Math.round((c.score || 0) * 100); const color = scoreColor(c.score); return `
${escapeHtml(c.wbs_code)} — ${escapeHtml(c.description)} ${isSelected ? 'Linked Candidate' : ''} ${scorePct}% match
`; }).join('') : '
No candidate matches found in this discipline baseline.
'}
`; }).join(''); } catch (err) { el.innerHTML = `
Error loading review queue: ${escapeHtml(err.message)}
`; } } async function accept(id) { try { await api(`/review/${id}/accept`, { method: 'POST' }); toast('Match confirmed — schedule updated'); load(); } catch(e){ toast(e.message); } } async function reject(id) { try { await api(`/review/${id}/reject`, { method: 'POST' }); toast('Match rejected'); load(); } catch(e){ toast(e.message); } } async function flagNew(id) { try { await api(`/review/${id}/flag-new`, { method: 'POST' }); toast('Flagged as new activity'); load(); } catch(e){ toast(e.message); } } async function reassign(matchId, activityId) { try { await api(`/review/${matchId}/reassign`, { method: 'POST', body: JSON.stringify({ activity_id: activityId }) }); toast('Reassigned candidate — click Accept to confirm'); load(); } catch(e){ toast(e.message); } } document.getElementById('statusFilter').onchange = load; document.getElementById('refresh').onclick = load; load();
